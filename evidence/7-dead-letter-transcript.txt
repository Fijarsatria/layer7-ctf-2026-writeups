//        mode 1: write user -> (header[0x10]+off)
//        mode 2: read  (ctx[0x38]+off) -> user   (bound ctx[0x40])
//      access bound is header[8]; the bitmap overlaps the header at
//      page+0x180, so we can OR bits into header[8].

//

//   page+0x100 : bitmap   page+0x180 : header   page+0x200 : body

static int page_read(uint32_t off, void *dst, uint32_t len) {
